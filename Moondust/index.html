Moondust Download moondust.lua · GitHub

A tiny package manager for Lua. Drop in one file, then require packages by name.

curl -O https://thegamingmahi.github.io/Moondust/moondust.lua
local moondust = require("moondust")
local json = moondust("alice/json")        -- latest
local old  = moondust("alice/json@1.0.0")  -- exact version
